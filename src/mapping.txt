# Format: <version-dir>  <uname -r>
# Example:
# 7.0              7.0.0-15-generic
# 6.12             6.12.0-10-generic

# dir              # kernel version string
# 7.3 directory is compatible up to 7.0
7.3                7.0
7.3                7.1
7.3                7.2
7.3                7.3
